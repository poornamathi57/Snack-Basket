// PostgreSQL Table Definitions for Snack Basket

-- 1. PROFILES & ROLES
CREATE TYPE user_role AS ENUM ('customer', 'vendor', 'admin', 'delivery');

CREATE TABLE profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  full_name TEXT NOT NULL,
  phone TEXT UNIQUE NOT NULL,
  role user_role DEFAULT 'customer',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. VENDORS (RESTAURANTS)
CREATE TABLE vendors (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  owner_id UUID REFERENCES profiles(id),
  restaurant_name TEXT NOT NULL,
  area_locality TEXT NOT NULL,
  address TEXT NOT NULL,
  fssai_license TEXT NOT NULL,
  commission_rate NUMERIC(4,2) DEFAULT 15.00,
  is_verified BOOLEAN DEFAULT false,
  is_active BOOLEAN DEFAULT true,
  rating NUMERIC(2,1) DEFAULT 4.5,
  banner_url TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. PRODUCTS (FOOD MENU ITEMS)
CREATE TABLE products (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  vendor_id UUID REFERENCES vendors(id) ON DELETE CASCADE,
  category_name TEXT NOT NULL, -- Puffs, Rolls, Momos, Waffles, etc.
  name TEXT NOT NULL,
  description TEXT,
  price NUMERIC(8,2) NOT NULL,
  image_url TEXT,
  is_veg BOOLEAN DEFAULT true,
  is_available BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. ORDERS & COMMISSIONS
CREATE TABLE orders (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  customer_id UUID REFERENCES profiles(id),
  vendor_id UUID REFERENCES vendors(id),
  subtotal NUMERIC(8,2) NOT NULL,
  delivery_fee NUMERIC(8,2) DEFAULT 35.00,
  tax_amount NUMERIC(8,2) NOT NULL,
  total_amount NUMERIC(8,2) NOT NULL,
  vendor_commission NUMERIC(8,2) NOT NULL,
  status TEXT CHECK (status IN ('placed', 'preparing', 'out_for_delivery', 'delivered', 'cancelled')) DEFAULT 'placed',
  payment_status TEXT DEFAULT 'pending',
  delivery_address TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- ROW LEVEL SECURITY (RLS) POLICIES
ALTER TABLE orders ENABLE ROW LEVEL SECURITY;

-- Customers can view only their own orders
CREATE POLICY customer_view_own_orders ON orders 
  FOR SELECT USING (auth.uid() = customer_id);

-- Vendors can view orders belonging to their restaurant
CREATE POLICY vendor_view_own_orders ON orders 
  FOR SELECT USING (
    auth.uid() IN (SELECT owner_id FROM vendors WHERE id = orders.vendor_id)
  );
